import React, { useState } from 'react';
import { 
  X, 
  MapPin, 
  Calendar, 
  Coins, 
  Users, 
  Building, 
  CheckCircle2, 
  Clock, 
  AlertCircle, 
  Check, 
  Printer, 
  Layers, 
  FileText,
  Sliders,
  ArrowRight,
  Sparkles,
  FileCheck2,
  Share2,
  Edit3
} from 'lucide-react';
import { Project, ProjectStatus, ProjectDocument } from '../types';
import { SearchableSelect } from './SearchableSelect';
import { ProjectPrintTemplate } from './ProjectPrintTemplate';
import { ProjectTimelineView } from './ProjectTimelineView';
import { ProjectDocumentsTab } from './ProjectDocumentsTab';
import { getDefaultProjectDocuments } from '../data/projectDocuments';

interface ProjectDetailModalProps {
  project: Project | null;
  onClose: () => void;
  onUpdateStatus: (id: number, status: ProjectStatus, progress?: number) => void;
  onEditProject?: (project: Project) => void;
  isStandalone?: boolean;
  onUpdateDocuments?: (projectId: number, documents: ProjectDocument[]) => void;
  onShowToast?: (message: string, type?: 'success' | 'error' | 'warning' | 'info') => void;
}

export const ProjectDetailModal: React.FC<ProjectDetailModalProps> = ({
  project,
  onClose,
  onUpdateStatus,
  onEditProject,
  isStandalone = false,
  onUpdateDocuments,
  onShowToast,
}) => {
  if (!project) return null;

  const [activeTab, setActiveTab] = useState<'info' | 'activities' | 'timeline' | 'documents' | 'workflow'>('info');
  const [currentProgress, setCurrentProgress] = useState<number>(project.progress_percentage);
  const [newStatus, setNewStatus] = useState<ProjectStatus>(project.status);
  const [isSaving, setIsSaving] = useState(false);
  const [isPrintOpen, setIsPrintOpen] = useState(false);

  // Documents state synchronized with project
  const [documents, setDocuments] = useState<ProjectDocument[]>(() => {
    return getDefaultProjectDocuments(project);
  });

  React.useEffect(() => {
    setDocuments(getDefaultProjectDocuments(project));
  }, [project.id, project.documents]);

  const handleUpdateDocuments = (updatedDocs: ProjectDocument[]) => {
    setDocuments(updatedDocs);
    if (onUpdateDocuments) {
      onUpdateDocuments(project.id, updatedDocs);
    }
  };

  const handleSaveWorkflow = () => {
    setIsSaving(true);
    onUpdateStatus(project.id, newStatus, currentProgress);
    setTimeout(() => {
      setIsSaving(false);
    }, 400);
  };

  const content = (
    <div className={`bg-white dark:bg-slate-900 rounded-2xl overflow-hidden flex flex-col text-slate-900 dark:text-white ${
      isStandalone 
        ? 'w-full shadow-lg border border-slate-200 dark:border-slate-800' 
        : 'max-w-4xl w-full max-h-[92vh] border border-amber-500/20 shadow-2xl'
    }`}>
      {/* Government Header with Navy Gradient & Gold Trim */}
      <div className="bg-gradient-to-r from-[#071320] via-[#0d2238] to-[#153454] text-white p-5 sm:p-6 border-b border-amber-500/30 flex items-start justify-between relative overflow-hidden">
        <div className="h-1 w-full bg-gradient-to-r from-amber-600 via-amber-400 to-amber-600 absolute top-0 left-0"></div>
        
        <div className="space-y-1.5 pr-2">
          <div className="flex flex-wrap items-center gap-2">
            <span className="bg-amber-400 text-slate-950 text-xs font-mono font-black px-2.5 py-0.5 rounded shadow-xs">
              {project.code}
            </span>
            <span className="text-amber-200 text-xs font-mono">نموذج رقم: {project.form_number}</span>
            <span className="text-slate-400 text-xs">• وثيقة رسمية معتمدة</span>
          </div>

          <h2 className="text-base sm:text-lg font-black leading-snug text-white">
            {project.project_name}
          </h2>

          <div className="text-xs text-slate-300 flex flex-wrap items-center gap-2">
            <span className="text-amber-300 font-semibold">{project.creator_entity}</span>
            <span>•</span>
            <span>{project.program}</span>
          </div>
        </div>

        <button
          onClick={onClose}
          className="text-slate-400 hover:text-white p-1.5 rounded-xl hover:bg-white/10 transition cursor-pointer flex items-center gap-1.5 text-xs font-bold"
          aria-label={isStandalone ? "العودة للخلف" : "إغلاق"}
        >
          {isStandalone ? (
            <>
              <span>العودة للمشاريع</span>
              <ArrowRight className="w-4 h-4 text-amber-400" />
            </>
          ) : (
            <X className="w-5 h-5" />
          )}
        </button>
      </div>

      {/* Tab Navigation Ribbon */}
      <div className="flex items-center gap-2 sm:gap-6 px-6 border-b border-slate-200 dark:border-slate-800 bg-slate-50 dark:bg-slate-900/60 text-xs font-bold text-slate-600 dark:text-slate-300 overflow-x-auto">
        <button
          onClick={() => setActiveTab('info')}
          className={`py-3.5 border-b-2 transition whitespace-nowrap cursor-pointer ${
            activeTab === 'info'
              ? 'border-amber-500 text-amber-600 dark:text-amber-400 font-black'
              : 'border-transparent hover:text-slate-900 dark:hover:text-white'
          }`}
        >
          بيانات وبطاقة المشروع
        </button>
        <button
          onClick={() => setActiveTab('activities')}
          className={`py-3.5 border-b-2 transition whitespace-nowrap cursor-pointer ${
            activeTab === 'activities'
              ? 'border-amber-500 text-amber-600 dark:text-amber-400 font-black'
              : 'border-transparent hover:text-slate-900 dark:hover:text-white'
          }`}
        >
          الأنشطة والتدخلات ({project.activities.length})
        </button>
        <button
          onClick={() => setActiveTab('timeline')}
          className={`py-3.5 border-b-2 transition whitespace-nowrap cursor-pointer flex items-center gap-1.5 ${
            activeTab === 'timeline'
              ? 'border-amber-500 text-amber-600 dark:text-amber-400 font-black'
              : 'border-transparent hover:text-slate-900 dark:hover:text-white'
          }`}
        >
          <Calendar className="w-3.5 h-3.5 text-amber-500" />
          <span>المخطط الزمني والمعالم (Timeline)</span>
        </button>
        <button
          onClick={() => setActiveTab('documents')}
          className={`py-3.5 border-b-2 transition whitespace-nowrap cursor-pointer flex items-center gap-1.5 ${
            activeTab === 'documents'
              ? 'border-amber-500 text-amber-600 dark:text-amber-400 font-black'
              : 'border-transparent hover:text-slate-900 dark:hover:text-white'
          }`}
        >
          <FileText className="w-3.5 h-3.5 text-amber-500" />
          <span>الوثائق والمستندات (Documents)</span>
          <span className={`px-1.5 py-0.5 rounded-full text-[10px] font-mono font-bold ${
            activeTab === 'documents'
              ? 'bg-amber-500 text-slate-950'
              : 'bg-slate-200 dark:bg-slate-800 text-slate-600 dark:text-slate-400'
          }`}>
            {documents.length}
          </span>
        </button>
        <button
          onClick={() => setActiveTab('workflow')}
          className={`py-3.5 border-b-2 transition whitespace-nowrap cursor-pointer ${
            activeTab === 'workflow'
              ? 'border-amber-500 text-amber-600 dark:text-amber-400 font-black'
              : 'border-transparent hover:text-slate-900 dark:hover:text-white'
          }`}
        >
          مسار الاعتماد ونسبة الإنجاز
        </button>
      </div>

      {/* Scrollable Body */}
      <div className={`p-5 sm:p-6 ${isStandalone ? '' : 'overflow-y-auto'} space-y-6 flex-1 text-xs`}>
          {activeTab === 'info' && (
            <div className="space-y-6">
              {/* Financial & Field Metrics */}
              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3.5">
                <div className="bg-slate-50 dark:bg-slate-800/60 p-4 rounded-xl border border-slate-200 dark:border-slate-800">
                  <div className="text-slate-500 dark:text-slate-400 mb-1 flex items-center gap-1.5">
                    <Coins className="w-4 h-4 text-amber-500" />
                    <span>الموازنة التقديرية</span>
                  </div>
                  <div className="text-base font-black font-mono text-slate-900 dark:text-white">
                    ${project.estimated_cost.toLocaleString()}
                  </div>
                </div>

                <div className="bg-slate-50 dark:bg-slate-800/60 p-4 rounded-xl border border-slate-200 dark:border-slate-800">
                  <div className="text-slate-500 dark:text-slate-400 mb-1 flex items-center gap-1.5">
                    <Coins className="w-4 h-4 text-emerald-500" />
                    <span>المنصرف الفعلي</span>
                  </div>
                  <div className="text-base font-black font-mono text-emerald-600 dark:text-emerald-400">
                    ${project.actual_spent.toLocaleString()}
                  </div>
                </div>

                <div className="bg-slate-50 dark:bg-slate-800/60 p-4 rounded-xl border border-slate-200 dark:border-slate-800">
                  <div className="text-slate-500 dark:text-slate-400 mb-1 flex items-center gap-1.5">
                    <Users className="w-4 h-4 text-purple-500" />
                    <span>المستفيدون المستهدفون</span>
                  </div>
                  <div className="text-base font-black font-mono text-slate-900 dark:text-white">
                    {project.beneficiaries_count.toLocaleString()}
                  </div>
                </div>

                <div className="bg-slate-50 dark:bg-slate-800/60 p-4 rounded-xl border border-slate-200 dark:border-slate-800">
                  <div className="text-slate-500 dark:text-slate-400 mb-1 flex items-center gap-1.5">
                    <MapPin className="w-4 h-4 text-rose-500" />
                    <span>النطاق الجغرافي</span>
                  </div>
                  <div className="text-sm font-bold text-slate-900 dark:text-white">
                    {project.governorate} • {project.directorate}
                  </div>
                </div>
              </div>

              {/* Description */}
              <div className="bg-white dark:bg-slate-800/40 p-4 rounded-xl border border-slate-200 dark:border-slate-800 space-y-1.5">
                <h4 className="font-bold text-slate-900 dark:text-white text-xs">وصف المشروع وخلفيته الفنية</h4>
                <p className="text-slate-600 dark:text-slate-300 leading-relaxed">
                  {project.description || 'مشروع تنموي معتمد ضمن الخطة الوطنية الشاملة.'}
                </p>
              </div>

              {/* Strategic Goals List */}
              <div className="bg-white dark:bg-slate-800/40 p-4 rounded-xl border border-slate-200 dark:border-slate-800 space-y-3">
                <h4 className="font-bold text-slate-900 dark:text-white text-xs">الأهداف الاستراتيجية ومؤشرات الأثر</h4>
                <div className="space-y-2">
                  {project.objectives && project.objectives.length > 0 ? (
                    project.objectives.map((obj, i) => (
                      <div key={i} className="flex items-start gap-2 text-slate-700 dark:text-slate-300">
                        <CheckCircle2 className="w-4 h-4 text-emerald-500 shrink-0 mt-0.5" />
                        <span>{obj}</span>
                      </div>
                    ))
                  ) : (
                    <p className="text-slate-400">لم تحدد أهداف خاصة</p>
                  )}
                </div>
              </div>

              {/* Administrative Details Table */}
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div className="bg-slate-50 dark:bg-slate-800/60 p-4 rounded-xl border border-slate-200 dark:border-slate-800 space-y-2">
                  <h4 className="font-bold text-slate-900 dark:text-white flex items-center gap-2">
                    <Calendar className="w-4 h-4 text-amber-500" />
                    <span>المواعيد والجدول الزمني المخطط</span>
                  </h4>
                  <div className="flex justify-between text-slate-600 dark:text-slate-300">
                    <span>تاريخ البدء:</span>
                    <span className="font-mono font-bold">{project.start_date_gregorian}</span>
                  </div>
                  <div className="flex justify-between text-slate-600 dark:text-slate-300">
                    <span>تاريخ الانتهاء المخطط:</span>
                    <span className="font-mono font-bold">{project.end_date_gregorian}</span>
                  </div>
                  <div className="flex justify-between text-slate-600 dark:text-slate-300">
                    <span>تاريخ القيد بالمنظومة:</span>
                    <span className="font-mono">{project.created_at}</span>
                  </div>
                </div>

                <div className="bg-slate-50 dark:bg-slate-800/60 p-4 rounded-xl border border-slate-200 dark:border-slate-800 space-y-2">
                  <h4 className="font-bold text-slate-900 dark:text-white flex items-center gap-2">
                    <Building className="w-4 h-4 text-amber-500" />
                    <span>الهيكل الإداري والجهات الشريكة</span>
                  </h4>
                  <div className="flex justify-between text-slate-600 dark:text-slate-300">
                    <span>الجهة المنشئة:</span>
                    <span className="font-semibold text-slate-900 dark:text-white">{project.creator_entity}</span>
                  </div>
                  <div className="flex justify-between text-slate-600 dark:text-slate-300">
                    <span>البرنامج الرئيسي:</span>
                    <span>{project.program}</span>
                  </div>
                  <div className="flex justify-between text-slate-600 dark:text-slate-300">
                    <span>القطاع والمجال الفرعي:</span>
                    <span>{project.domain} • {project.subdomain}</span>
                  </div>
                </div>
              </div>
            </div>
          )}

          {activeTab === 'activities' && (
            <div className="space-y-4">
              <div className="flex items-center justify-between pb-2 border-b border-slate-200 dark:border-slate-800">
                <h4 className="font-bold text-slate-900 dark:text-white text-sm">
                  جدول الأنشطة التنفيذية والتمهيدية الميدانية
                </h4>
                <span className="text-slate-500 dark:text-slate-400 font-mono">
                  {project.activities.length} أنشطة
                </span>
              </div>

              {project.activities.length === 0 ? (
                <div className="text-center py-12 bg-slate-50 dark:bg-slate-800/50 rounded-xl border border-slate-200 dark:border-slate-800 text-slate-400">
                  لم يتم إدراج أنشطة تفصيلية لهذا المشروع بعد.
                </div>
              ) : (
                <div className="bg-white dark:bg-slate-900 rounded-xl border border-slate-200 dark:border-slate-800 overflow-hidden">
                  <div className="overflow-x-auto">
                    <table className="w-full text-right text-xs">
                      <thead className="bg-slate-50 dark:bg-slate-800/80 text-slate-600 dark:text-slate-300 font-bold border-b border-slate-200 dark:border-slate-700">
                        <tr>
                          <th className="py-3 px-3.5 whitespace-nowrap">نوع النشاط</th>
                          <th className="py-3 px-3.5 whitespace-nowrap">عنوان النشاط</th>
                          <th className="py-3 px-3.5 whitespace-nowrap">الجهة المنفذة</th>
                          <th className="py-3 px-3.5 whitespace-nowrap">تاريخ البدء</th>
                          <th className="py-3 px-3.5 whitespace-nowrap">تاريخ الانتهاء</th>
                          <th className="py-3 px-3.5 whitespace-nowrap">التكلفة التقديرية</th>
                          <th className="py-3 px-3.5 whitespace-nowrap">نسبة الإنجاز</th>
                        </tr>
                      </thead>
                      <tbody className="divide-y divide-slate-100 dark:divide-slate-800">
                        {project.activities.map((act) => (
                          <tr key={act.id} className="hover:bg-slate-50 dark:hover:bg-slate-800/40 transition">
                            <td className="py-3 px-3.5 whitespace-nowrap">
                              <span className={`px-2 py-0.5 rounded text-[10px] font-bold ${
                                act.type === 'preliminary' 
                                  ? 'bg-amber-500/15 text-amber-700 dark:text-amber-400 border border-amber-500/30' 
                                  : 'bg-blue-500/15 text-blue-700 dark:text-blue-400 border border-blue-500/30'
                              }`}>
                                {act.type === 'preliminary' ? 'نشاط تمهيدي' : 'نشاط تنفيذي'}
                              </span>
                            </td>
                            <td className="py-3 px-3.5 whitespace-nowrap font-bold text-slate-900 dark:text-white">
                              {act.title}
                            </td>
                            <td className="py-3 px-3.5 whitespace-nowrap text-slate-600 dark:text-slate-300">
                              {act.assignedEntity}
                            </td>
                            <td className="py-3 px-3.5 whitespace-nowrap font-mono text-slate-500 dark:text-slate-400">
                              {act.startDate}
                            </td>
                            <td className="py-3 px-3.5 whitespace-nowrap font-mono text-slate-500 dark:text-slate-400">
                              {act.endDate}
                            </td>
                            <td className="py-3 px-3.5 whitespace-nowrap font-mono font-bold text-slate-900 dark:text-white">
                              ${act.cost.toLocaleString()}
                            </td>
                            <td className="py-3 px-3.5 whitespace-nowrap">
                              <div className="flex items-center gap-2">
                                <div className="w-16 bg-slate-100 dark:bg-slate-800 h-2 rounded-full overflow-hidden">
                                  <div 
                                    className="bg-emerald-500 h-full rounded-full"
                                    style={{ width: `${act.progress}%` }}
                                  />
                                </div>
                                <span className="font-mono font-bold text-[11px] text-slate-700 dark:text-slate-300">
                                  {act.progress}%
                                </span>
                              </div>
                            </td>
                          </tr>
                        ))}
                      </tbody>
                    </table>
                  </div>
                </div>
              )}
            </div>
          )}

          {activeTab === 'timeline' && (
            <ProjectTimelineView project={project} />
          )}

          {activeTab === 'documents' && (
            <ProjectDocumentsTab
              project={project}
              documents={documents}
              onUpdateDocuments={handleUpdateDocuments}
              onShowToast={onShowToast}
            />
          )}

          {activeTab === 'workflow' && (
            <div className="space-y-6">
              {/* Approval Stages Stepper */}
              <div className="bg-slate-50 dark:bg-slate-800/60 p-5 rounded-2xl border border-slate-200 dark:border-slate-800 space-y-4">
                <h4 className="font-bold text-slate-900 dark:text-white text-xs flex items-center gap-2">
                  <Sparkles className="w-4 h-4 text-amber-500" />
                  <span>دورة الاعتماد والمصادقة المؤسسية للمشروع</span>
                </h4>
                
                <div className="grid grid-cols-1 sm:grid-cols-4 gap-3 text-center text-xs">
                  <div className="bg-white dark:bg-slate-900 p-3 rounded-xl border border-emerald-500/40 shadow-xs">
                    <div className="text-emerald-600 dark:text-emerald-400 font-bold mb-1">1. إعداد المسودة</div>
                    <div className="text-[10px] text-slate-400">مكتمل وموثق</div>
                  </div>

                  <div className={`p-3 rounded-xl border ${
                    project.status !== 'draft'
                      ? 'bg-white dark:bg-slate-900 border-amber-500/40 shadow-xs text-amber-600 dark:text-amber-400 font-bold'
                      : 'bg-slate-100 dark:bg-slate-800/40 border-slate-200 dark:border-slate-700 text-slate-400'
                  }`}>
                    <div className="mb-1">2. المراجعة الفنية</div>
                    <div className="text-[10px] text-slate-400 font-normal">مطابقة الجداول</div>
                  </div>

                  <div className={`p-3 rounded-xl border ${
                    ['approved', 'in_execution', 'completed'].includes(project.status)
                      ? 'bg-white dark:bg-slate-900 border-emerald-500/40 shadow-xs text-emerald-600 dark:text-emerald-400 font-bold'
                      : 'bg-slate-100 dark:bg-slate-800/40 border-slate-200 dark:border-slate-700 text-slate-400'
                  }`}>
                    <div className="mb-1">3. الاعتماد والتمويل</div>
                    <div className="text-[10px] text-slate-400 font-normal">الأمانة العامة</div>
                  </div>

                  <div className={`p-3 rounded-xl border ${
                    project.status === 'completed'
                      ? 'bg-white dark:bg-slate-900 border-emerald-500/40 shadow-xs text-emerald-600 dark:text-emerald-400 font-bold'
                      : 'bg-slate-100 dark:bg-slate-800/40 border-slate-200 dark:border-slate-700 text-slate-400'
                  }`}>
                    <div className="mb-1">4. التسليم والإغلاق</div>
                    <div className="text-[10px] text-slate-400 font-normal">الاستلام النهائي</div>
                  </div>
                </div>
              </div>

              {/* Status & Progress Form */}
              <div className="bg-white dark:bg-slate-900 p-5 rounded-2xl border border-slate-200 dark:border-slate-800 space-y-4">
                <h4 className="font-bold text-slate-900 dark:text-white text-xs">
                  تحديث حالة المشروع ونسبة الإنجاز الميداني
                </h4>

                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-slate-700 dark:text-slate-300 font-semibold mb-1">
                      حالة المشروع المعتمدة:
                    </label>
                    <SearchableSelect
                      value={newStatus}
                      onChange={(val) => setNewStatus(val as ProjectStatus)}
                      placeholder="تحديد حالة المشروع..."
                      searchPlaceholder="ابحث في الحالات..."
                      options={[
                        { value: 'draft', label: 'مسودة أولية' },
                        { value: 'under_review', label: 'قيد المراجعة الفنية والمالية' },
                        { value: 'approved', label: 'معتمد رسمياً للتمويل' },
                        { value: 'in_execution', label: 'قيد التنفيذ الميداني' },
                        { value: 'completed', label: 'مكتمل ومنجز بنجاح' },
                        { value: 'on_hold', label: 'معلق مؤقتاً' },
                      ]}
                    />
                  </div>

                  <div>
                    <label className="block text-slate-700 dark:text-slate-300 font-semibold mb-1">
                      نسبة الإنجاز الميداني: <span className="text-amber-600 dark:text-amber-400 font-black font-mono">{currentProgress}%</span>
                    </label>
                    <input
                      type="range"
                      min="0"
                      max="100"
                      value={currentProgress}
                      onChange={(e) => setCurrentProgress(Number(e.target.value))}
                      className="w-full h-2 bg-slate-200 dark:bg-slate-700 rounded-lg appearance-none cursor-pointer accent-amber-500 mt-3"
                    />
                  </div>
                </div>

                <div className="flex justify-end pt-2">
                  <button
                    onClick={handleSaveWorkflow}
                    disabled={isSaving}
                    className="flex items-center gap-2 bg-gradient-to-r from-amber-400 to-amber-500 hover:from-amber-300 hover:to-amber-400 text-slate-950 font-black px-5 py-2.5 rounded-xl text-xs shadow-md shadow-amber-500/20 transition cursor-pointer disabled:opacity-50"
                  >
                    <Check className="w-4 h-4" />
                    <span>{isSaving ? 'جارٍ الحفظ...' : 'حفظ التحديثات الرسمية'}</span>
                  </button>
                </div>
              </div>
            </div>
          )}
        </div>

        {/* Modal Footer */}
        <div className="bg-slate-50 dark:bg-slate-800/80 p-4 border-t border-slate-200 dark:border-slate-800 flex items-center justify-between text-xs">
          <div className="text-slate-500 dark:text-slate-400 flex items-center gap-1.5">
            <span>المرحلة:</span>
            <span className="font-bold text-slate-900 dark:text-white">{project.current_approval_stage}</span>
          </div>

          <div className="flex items-center gap-2">
            {onEditProject && (
              <button
                onClick={() => {
                  onClose();
                  onEditProject(project);
                }}
                className="px-3.5 py-2 bg-[#c9a961]/15 border border-[#c9a961]/40 rounded-xl hover:bg-[#c9a961]/25 text-[#d4b876] font-bold flex items-center gap-1.5 transition cursor-pointer"
              >
                <Edit3 className="w-4 h-4 text-[#c9a961]" />
                <span>تعديل بيانات المشروع</span>
              </button>
            )}
            <button
              onClick={() => setIsPrintOpen(true)}
              className="px-3.5 py-2 border border-slate-200 dark:border-slate-700 rounded-xl hover:bg-white dark:hover:bg-slate-700 text-slate-700 dark:text-slate-200 font-bold flex items-center gap-1.5 transition cursor-pointer"
            >
              <Printer className="w-4 h-4 text-amber-500" />
              <span>طباعة بطاقة المشروع</span>
            </button>
            <button
              onClick={onClose}
              className="px-4 py-2 bg-slate-200 dark:bg-slate-700 hover:bg-slate-300 dark:hover:bg-slate-600 rounded-xl text-slate-800 dark:text-slate-200 font-bold transition cursor-pointer"
            >
              إغلاق
            </button>
          </div>
        </div>
      </div>
    );

  if (isStandalone) {
    return (
      <div className="w-full space-y-6">
        {content}
        <ProjectPrintTemplate project={project} isOpen={isPrintOpen} onClose={() => setIsPrintOpen(false)} />
      </div>
    );
  }

  return (
    <div className="fixed inset-0 z-50 overflow-y-auto bg-slate-950/70 backdrop-blur-xs flex items-center justify-center p-3 sm:p-5 animate-in fade-in duration-200">
      {content}
      <ProjectPrintTemplate project={project} isOpen={isPrintOpen} onClose={() => setIsPrintOpen(false)} />
    </div>
  );
};
